/-! Uniform encoding (B): `abbrev n : Prop := S` (statement) + `n.holds` (justification) -/
axiom Character : Type
axiom Game : Type
axiom Cynder : Character
axiom newSpyro : Game
axiom HasDarkPast : Character → Prop
axiom IsPG : Game → Prop
axiom GoodAddition : Character → Game → Prop
axiom age : Character → Nat

abbrev cynder_dark : Prop := HasDarkPast Cynder
axiom cynder_dark.holds : cynder_dark                       -- ASSUMPTION
abbrev spyro_pg : Prop := IsPG newSpyro
axiom spyro_pg.holds : spyro_pg                             -- ASSUMPTION
abbrev pg_no_dark : Prop := ∀ g c, IsPG g → HasDarkPast c → ¬ GoodAddition c g
axiom pg_no_dark.holds : pg_no_dark                         -- ASSUMPTION
abbrev cynder_age : Prop := age Cynder = 17
axiom cynder_age.holds : cynder_age
abbrev cynder_adult_bound : Prop := age Cynder ≤ 20
axiom cynder_adult_bound.holds : cynder_adult_bound

abbrev cynder_not_good : Prop := ¬ GoodAddition Cynder newSpyro   -- GOAL statement
-- proofs in different styles:
theorem cynder_not_good.holds : cynder_not_good := pg_no_dark.holds _ _ spyro_pg.holds cynder_dark.holds
example : cynder_not_good := by apply pg_no_dark.holds <;> first | exact spyro_pg.holds | exact cynder_dark.holds
example : cynder_not_good := by intro h; exact pg_no_dark.holds _ _ spyro_pg.holds cynder_dark.holds h
example : cynder_not_good := by simp [pg_no_dark.holds, spyro_pg.holds, cynder_dark.holds]
example : cynder_not_good := by grind [pg_no_dark.holds, spyro_pg.holds, cynder_dark.holds]
example : age Cynder + 1 = 18 := by rw [cynder_age.holds]
example : age Cynder + 1 = 18 := by simp [cynder_age.holds]
example : age Cynder < 25 := by have := cynder_adult_bound.holds; omega
example : age Cynder < 25 := by omega   -- without the hypothesis: should fail
#print axioms cynder_not_good.holds
