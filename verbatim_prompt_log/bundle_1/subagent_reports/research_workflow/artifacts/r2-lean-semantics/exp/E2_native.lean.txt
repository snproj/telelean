-- a 'false' implemented_by: compiled code disagrees with the logic
def realF (n : Nat) : Bool := false
@[implemented_by realF] def logicalF (n : Nat) : Bool := true
-- logically logicalF 0 = true; compiled = false
theorem logical_ok : logicalF 0 = true := rfl
#print axioms logical_ok
theorem lie : logicalF 0 = false := by native_decide
#print axioms lie
theorem contradiction : False := by
  have h1 := logical_ok
  have h2 := lie
  rw [h1] at h2
  exact Bool.noConfusion h2
#print axioms contradiction
