import Model

theorem telelean_independence :
    Exists fun (Character : Type) =>
      Exists fun (Game : Type) =>
        Exists fun (cynder : Character) =>
          Exists fun (newSpyro : Game) =>
            Exists fun (HasDarkPast : Character → Prop) =>
              Exists fun (IsPG : Game → Prop) =>
                Exists fun (AppearsIn : Character → Game → Prop) =>
                  Exists fun (GoodAdditionTo : Character → Game → Prop) =>
                    And (HasDarkPast cynder)
                      (And (IsPG newSpyro)
                        (And
                          (∀ (g : Game) (c : Character),
                            IsPG g → HasDarkPast c → Not (AppearsIn c g))
                          (Not (Not (GoodAdditionTo cynder newSpyro))))) := by
  -- countermodel: "good addition" holds of everything, nobody appears anywhere
  exact ⟨Unit, Unit, (), (), fun _ => True, fun _ => True, fun _ _ => False, fun _ _ => True,
    trivial, trivial, fun _ _ _ _ h => h, fun h => h trivial⟩
