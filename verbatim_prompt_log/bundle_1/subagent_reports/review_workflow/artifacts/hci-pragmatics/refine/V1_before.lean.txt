-- Design's default encoding (doc 03 §4): uninterpreted vocabulary + Prop axioms.
axiom Character : Type
axiom Game : Type
axiom cynder : Character
axiom newSpyro : Game
axiom HasDarkPast : Character → Prop
axiom IsPG : Game → Prop
axiom CanAppearIn : Character → Game → Prop
axiom GoodAddition : Character → Game → Prop
axiom cynder_dark_past : HasDarkPast cynder                                   -- h1
axiom new_spyro_pg : IsPG newSpyro                                            -- h2
axiom pg_no_dark_past : ∀ g c, IsPG g → HasDarkPast c → ¬ CanAppearIn c g     -- h3
axiom good_needs_appear : ∀ c g, GoodAddition c g → CanAppearIn c g           -- bridge
theorem cynder_not_good : ¬ GoodAddition cynder newSpyro :=
  fun h => pg_no_dark_past _ _ new_spyro_pg cynder_dark_past (good_needs_appear _ _ h)
#print axioms cynder_not_good
